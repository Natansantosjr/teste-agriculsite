# 📊 5. SCHEMAS (ENTRADA/SAÍDA)

## schemas.py

```python
from pydantic import BaseModel

class AgroInput(BaseModel):
    ndvi: float
    rainfall: float
    temperature: float
    soil_moisture: float
    area_ha: float
    declared_production: float

class PredictionOutput(BaseModel):
    estimated_production: float
    anomaly_score: int
    risk_level: str
```

# ⚙️ 6. LÓGICA DE NEGÓCIO

## prediction.py

```python
from app.model import production_model

def predict_production(data):
    features = [[
        data.ndvi,
        data.rainfall,
        data.temperature,
        data.soil_moisture,
        data.area_ha
    ]]

    return production_model.predict(features)[0]
```

---

## anomaly.py

```python
from app.model import anomaly_model

def detect_anomaly(estimated, declared):
    diff = abs(estimated - declared)

    score = anomaly_model.predict([[estimated, declared, diff]])[0]

    return score
```

# 🌐 7. API PRINCIPAL

## main.py

```python
from fastapi import FastAPI
from app.schemas import AgroInput, PredictionOutput
from app.services.prediction import predict_production
from app.services.anomaly import detect_anomaly

app = FastAPI(title="Agro AI API")

@app.get("/")
def root():
    return {"status": "running"}

@app.post("/predict", response_model=PredictionOutput)
def predict(data: AgroInput):

    estimated = predict_production(data)

    anomaly = detect_anomaly(estimated, data.declared_production)

    if anomaly == -1:
        risk = "HIGH"
    else:
        risk = "LOW"

    return {
        "estimated_production": estimated,
        "anomaly_score": anomaly,
        "risk_level": risk
    }
```
